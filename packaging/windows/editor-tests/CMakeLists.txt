cmake_minimum_required(VERSION 3.24)
project(PadNativeEditorTests LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This fixture targets native Windows MSVC")
endif()
# Build the ordinary project unchanged, then assemble a separate executable
# from the same sources/dependencies, replacing only its main entry point.
set(PAD_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(USE_QT6_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(PORTABLE_WIN ON CACHE BOOL "" FORCE)
set(CHECK_FOR_UPDATES OFF CACHE BOOL "" FORCE)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Network Concurrent LinguistTools Test)
add_subdirectory("${PAD_ROOT}" app EXCLUDE_FROM_ALL)
get_target_property(app_sources antimicrox SOURCES)
get_target_property(app_includes antimicrox INCLUDE_DIRECTORIES)
get_target_property(app_links antimicrox LINK_LIBRARIES)
get_directory_property(app_definitions DIRECTORY "${PAD_ROOT}" COMPILE_DEFINITIONS)
set(fixture_sources)
foreach(source IN LISTS app_sources)
    if(source MATCHES "(^|/)main\\.cpp$" OR source MATCHES "antimicrox_windows_icon\\.rc$")
        continue()
    endif()
    if(NOT IS_ABSOLUTE "${source}")
        set(source "${PAD_ROOT}/${source}")
    endif()
    list(APPEND fixture_sources "${source}")
endforeach()
add_executable(pad-editor-tests editor-controls.cpp ${fixture_sources})
set_target_properties(pad-editor-tests PROPERTIES AUTOMOC ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(pad-editor-tests PRIVATE ${app_includes} "${PAD_ROOT}" "${PAD_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}/app" "${CMAKE_CURRENT_BINARY_DIR}/app/src")
target_compile_definitions(pad-editor-tests PRIVATE ${app_definitions} WIN_PORTABLE_PACKAGE _CRT_SECURE_NO_WARNINGS)
target_link_libraries(pad-editor-tests PRIVATE ${app_links} Qt5::Test)
